Star Wars Character Finder

Click the button to find a character!

// 1. Retrieve elements from the DOM const contentDiv = document.getElementById('content'); const btn = document.getElementById('fetch-btn'); // 2. Function to get data from API async function getStarWarsCharacter(){ displayLoading(); // Generate a random ID (SWAPI has ~83 characters) const randomId = Math.floor(Math.random() * 83) + 1; try { const response = await fetch(`https://www.swapi.tech/api/people/${randomId}`); if (!response.ok) throw new Error("Character not found"); const data = await response.json(); const character = data.result.properties; // Fetch Homeworld Name (it comes as a URL) const homeWorldResponse = await fetch(character.homeworld); const homeWorldData = await homeWorldResponse.json(); const homeWorldName = homeWorldData.result.properties.name; // 3. Display info on the DOM displayCharacter(character, homeWorldName); } catch (error) { displayError(); } } function displayLoading() { contentDiv.innerHTML = `

Searching the galaxy...

`; } function displayCharacter(char, planet) { contentDiv.innerHTML = `

${char.name}

Height: ${char.height}

Gender: ${char.gender}

Birth Year: ${char.birth_year}

Home World: ${planet}

`; } function displayError() { contentDiv.innerHTML = `

Oh No! That person isn't in the records.

`; } // Event Listener btn.addEventListener('click', getStarWarsCharacter); body { background-color: #1a1a1a; color: #ffe81f; /* Star Wars Yellow */ font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; display: flex; justify-content: center; align-items: center; height: 100vh; margin: 0; } .container { background: #2b2b2b; padding: 2rem; border-radius: 10px; text-align: center; box-shadow: 0 10px 30px rgba(0,0,0,0.5); width: 350px; } button { background-color: #ffe81f; border: none; padding: 10px 20px; font-weight: bold; cursor: pointer; border-radius: 5px; margin-top: 20px; transition: transform 0.2s; } button:hover { transform: scale(1.05); } .loading i { font-size: 3rem; margin-bottom: 10px; } .error { color: #ff4d4d; }
