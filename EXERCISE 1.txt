// Using a Union Type to restrict 'lastAction' to specific string literals
type ActionType = 'increment' | 'decrement' | 'none';

export const Counter = () => {
  // TypeScript infers 'count' is a number because the initial value is 0
  const [count, setCount] = useState(0);
  
  // Explicitly typing the state using our ActionType union
  const [lastAction, setLastAction] = useState<ActionType>('none');

  const handleIncrement = (): void => {
    setCount(prev => prev + 1);
    setLastAction('increment');
  };

  const handleDecrement = (): void => {
    setCount(prev => prev - 1);
    setLastAction('decrement');
  };

  return (
    <div style={{ border: '1px solid #ccc', padding: '1rem', margin: '1rem 0' }}>
      <h2>Interactive Counter</h2>
      <p style={{ fontSize: '1.5rem', fontWeight: 'bold' }}>Count: {count}</p>
      <p>Last Action: <span style={{ color: 'blue' }}>{lastAction}</span></p>
      
      <button onClick={handleIncrement} style={{ marginRight: '0.5rem' }}>Increment</button>
      <button onClick={handleDecrement}>Decrement</button>
    </div>
  );

import { useState, useEffect } from 'react';

// 1. Define the structure of the JSONPlaceholder user object
interface User {
  id: number;
  name: string;
  email: string;
  company: {
    name: string;
  };
}

export const UserList = () => {
  // Typings for our states: users array, loading boolean, and error string/null
  const [users, setUsers] = useState<User[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const fetchUsers = async () => {
      try {
        setLoading(true);
        const response = await fetch('https://jsonplaceholder.typicode.com/users');
        
        if (!response.ok) {
          throw new Error('Failed to fetch user data.');
        }
        
        const data: User[] = await response.json();
        setUsers(data);
      } catch (err) {
        // Asserting error type safely since caught errors are of type 'unknown' natively
        if (err instanceof Error) {
          setError(err.message);
        } else {
          setError('An unexpected error occurred.');
        }
      } finally {
        setLoading(false);
      }
    };

    fetchUsers();
  }, []); // Empty dependency array ensures this runs once when the component mounts

  if (loading) return <p>Loading users...</p>;
  if (error) return <p style={{ color: 'red' }}>Error: {error}</p>;

  return (
    <div style={{ border: '1px solid #ccc', padding: '1rem', margin: '1rem 0' }}>
      <h2>Fetched User Directory</h2>
      <ul style={{ paddingLeft: '20px' }}>
        {users.map(user => (
          // TypeScript verifies 'user.id', 'user.name', etc. exist on our User interface
          <li key={user.id} style={{ marginBottom: '0.5rem' }}>
            <strong>{user.name}</strong> ({user.email}) - <em>{user.company.name}</em>
          </li>
        ))}
      </ul>
    </div>
  );
};
